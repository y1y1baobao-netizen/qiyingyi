Number（数值）

let qyy_age = 19;            // 整数
let qyy_price = 198.50;        // 浮点数
let qyy_big = 2e8;            // 科学计数法
let qyy_inf = Infinity;     // 无穷大
let qyy_nan = NaN;          // 非数字
结果：
2

String（字符串）

let qyy_name = "齐英伊";     // 双引号
let qyy_id = '24001010829';   // 单引号
let qyy_greet = `你好, ${qyy_name}`; // 模板字符串
结果：
3

Boolean（布尔）

let qyy_isStudent = true;
let qyy_pass = 59 >= 60;   // 表达式返回 false
结果：
4

Null（空）

let qyy_empt = null;  // 明确表示"没有值"
// typeof null 返回 "object" 是历史遗留 bug
结果：
5

Undefined（未定义）

let qyy_x;                // 声明但未赋值
let qyy_y = undefined;  // 显式赋值
结果：
6

Symbol（符号，ES6 新增）

let qyy_key1 = Symbol("id");
let qyy_key2 = Symbol("id");
// 每个 Symbol 唯一, 即使描述相同也不相等
结果：
补
