核心代码

<!-- 引入字体图标 CSS（含 @font-face 定义） -->
<link rel="stylesheet" href="cdn/fa-all.min.css">

<!-- 使用类名调用图标，像文字一样设置样式 -->
<i class="fa-solid fa-heart"></i>
<i class="fa-brands fa-github"></i>

/* 字体图标本质是文字，可用 CSS 控制 */
.qyy_icon-item i {
  font-size: 2.4rem;
  color: var(--qyy-primary);
}
